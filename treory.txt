6.1 функция центра кластера (задание 27)
def centre(cl):
    m = []
    for p in cl:
        s = sum(dist(p, p1) for p1 in cl)
        m.append([s, p])
    return min(m)[1]
6.2 функция простых делителей (задание 25)
def p(x):
    for i in range(2, int(x ** 0.5) + 1):
        if x % i == 0:
            return [i] + p(x // i)
    return [x]
6.3 функция поиска всех делителей (задание 25)
def div(x):
    h = []
    for i in range(1, int(x ** 0.5) + 1):
        if x % i == 0:
            h.append(i)
            h.append(x // i)
    return h
