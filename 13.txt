from ipaddress import *
0. Индекс адреса сети - "0"
Индекс широковещательного адреса - "-1"
1. ip_address() — один адрес
a = ip_address("192.168.1.10")
int(a) -> адрес как целое число (32 бита для IPv4)
str(a) -> "192.168.1.10"
bits = format(int(a), "032b") - выведи число в двоичном виде, и если оно короче 32 символов — допиши нулей слева, пока не станет ровно 32

2. ip_network() — сеть (адрес + маска)
net = ip_network("192.168.1.0/24", 0)
net = ip_network("192.168.1.0/255.255.255.0", 0)   # маска в десятичном виде (количество единиц) — тоже работает, всегда после пишем ",0"

3. net.network_address	адрес сети (первый адрес, все хостовые биты = 0)
net.netmask	маска в виде адреса, например 255.255.255.0
net.hostmask	инверсия маски, например 0.0.0.255
net.prefixlen	число единиц в маске (/24 → 24)
net.num_addresses	всего адресов в сети (2^(32-prefixlen))

4.
for ip in net:            # ВСЕ адреса сети, включая network_address и broadcast_address
    ...

for ip in net.hosts():    # только адреса УЗЛОВ — без network_address и broadcast_address
    ...
